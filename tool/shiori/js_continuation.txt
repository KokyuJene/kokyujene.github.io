      lavender: {

    // Load data from localStorage
    function loadData() {
      const savedDiaries = localStorage.getItem('shiori_diaries');
      const savedEvents = localStorage.getItem('shiori_events');
      const savedTheme = localStorage.getItem('shiori_theme');
      const savedFont = localStorage.getItem('shiori_font');
      
      if (savedDiaries) diaries = JSON.parse(savedDiaries);
      if (savedEvents) events = JSON.parse(savedEvents);
      if (savedTheme) applyThemeColors(JSON.parse(savedTheme));
      if (savedFont) {
        document.documentElement.style.setProperty('--font-family', savedFont);
        document.getElementById('fontSelector').value = savedFont;
      }
    }

    // Save data to localStorage
    function saveDiaries() {
      localStorage.setItem('shiori_diaries', JSON.stringify(diaries));
    }

    function saveEvents() {
      localStorage.setItem('shiori_events', JSON.stringify(events));
    }

    // Page switching
    function switchPage(pageName) {
      document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
      document.getElementById(pageName).classList.add('active');
      
      document.querySelectorAll('.nav-button').forEach(btn => btn.classList.remove('active'));
      document.querySelectorAll('.bottom-nav-button').forEach(btn => btn.classList.remove('active'));
      
      event.currentTarget.classList.add('active');
      
      if (pageName === 'diary') renderDiaries();
      if (pageName === 'calendar') {
        renderCalendar();
        renderEvents();
      }
    }

    // Diary functions
    function openEditor(id = null) {
      editingDiaryId = id;
      const modal = document.getElementById('editorModal');
      const editor = document.getElementById('diaryEditor');
      const title = modal.querySelector('.modal-title');
      
      if (id) {
        const diary = diaries.find(d => d.id === id);
        editor.innerHTML = diary.content;
        title.textContent = '日記を編集';
      } else {
        editor.innerHTML = '';
        title.textContent = '日記を書く';
      }
      
      modal.classList.add('active');
      editor.focus();
    }

    function closeEditor() {
      document.getElementById('editorModal').classList.remove('active');
      editingDiaryId = null;
    }

    function saveDiary() {
      const content = document.getElementById('diaryEditor').innerHTML;
      if (!content.trim()) {
        alert('内容を入力してください');
        return;
      }
      
      if (editingDiaryId) {
        const diary = diaries.find(d => d.id === editingDiaryId);
        diary.content = content;
        diary.updatedAt = new Date().toISOString();
      } else {
        diaries.unshift({
          id: Date.now(),
          content: content,
          createdAt: new Date().toISOString(),
          updatedAt: new Date().toISOString()
        });
      }
      
      saveDiaries();
      renderDiaries();
      closeEditor();
    }

    function renderDiaries() {
      const list = document.getElementById('diaryList');
      list.innerHTML = '';
      
      diaries.forEach(diary => {
        const card = document.createElement('div');
        card.className = 'diary-card';
        if (selectedItems.diary.has(diary.id)) card.classList.add('selected');
        
        const date = new Date(diary.createdAt);
        const dateStr = `${date.getFullYear()}年${date.getMonth() + 1}月${date.getDate()}日`;
        
        const preview = document.createElement('div');
        preview.className = 'diary-preview';
        preview.innerHTML = diary.content;
        
        card.innerHTML = `
          <div class="diary-card-checkbox" onclick="toggleDiarySelection(${diary.id}, event)"></div>
          <div class="diary-date">${dateStr}</div>
        `;
        card.appendChild(preview);
        
        card.onclick = (e) => {
          if (!selectionMode.diary && !e.target.classList.contains('diary-card-checkbox')) {
            openEditor(diary.id);
          }
        };
        
        list.appendChild(card);
      });
      
      if (diaries.length === 0) {
        list.innerHTML = '<p style="text-align:center; color:var(--text-tertiary); padding:40px;">まだ日記がありません</p>';
      }
    }

    // Selection mode
    function toggleSelectionMode(type) {
      selectionMode[type] = !selectionMode[type];
      selectedItems[type].clear();
      
      const container = document.getElementById(type === 'diary' ? 'diary' : 'calendar');
      if (selectionMode[type]) {
        container.classList.add('selection-mode');
      } else {
        container.classList.remove('selection-mode');
      }
      
      const bar = document.getElementById(type === 'diary' ? 'diarySelectionBar' : 'eventSelectionBar');
      bar.classList.toggle('active', selectionMode[type]);
      
      if (type === 'diary') renderDiaries();
      if (type === 'event') renderEvents();
      updateSelectionCount(type);
    }

    function toggleDiarySelection(id, event) {
      event.stopPropagation();
      if (selectedItems.diary.has(id)) {
        selectedItems.diary.delete(id);
      } else {
        selectedItems.diary.add(id);
      }
      renderDiaries();
      updateSelectionCount('diary');
    }

    function toggleEventSelection(id, event) {
      event.stopPropagation();
      if (selectedItems.event.has(id)) {
        selectedItems.event.delete(id);
      } else {
        selectedItems.event.add(id);
      }
      renderEvents();
      updateSelectionCount('event');
    }

    function updateSelectionCount(type) {
      const countEl = document.getElementById(type === 'diary' ? 'diarySelectedCount' : 'eventSelectedCount');
      countEl.textContent = `${selectedItems[type].size}件選択中`;
    }

    function deleteSelectedDiaries() {
      if (selectedItems.diary.size === 0) return;
      if (!confirm(`${selectedItems.diary.size}件の日記を削除しますか?`)) return;
      
      diaries = diaries.filter(d => !selectedItems.diary.has(d.id));
      saveDiaries();
      cancelSelection('diary');
    }

    function deleteSelectedEvents() {
      if (selectedItems.event.size === 0) return;
      if (!confirm(`${selectedItems.event.size}件の予定を削除しますか?`)) return;
      
      events = events.filter(e => !selectedItems.event.has(e.id));
      saveEvents();
      cancelSelection('event');
    }

    function cancelSelection(type) {
      toggleSelectionMode(type);
    }

    // Editor commands
    function execCmd(command, value = null) {
      document.execCommand(command, false, value);
    }

    function insertHeading(tag) {
      const selection = window.getSelection();
      if (selection.rangeCount > 0) {
        const range = selection.getRangeAt(0);
        const heading = document.createElement(tag);
        heading.textContent = selection.toString() || `見出し ${tag.toUpperCase()}`;
        range.deleteContents();
        range.insertNode(heading);
      }
    }

    // Calendar functions
    function renderCalendar() {
      const grid = document.getElementById('calendarGrid');
      const monthEl = document.getElementById('currentMonth');
      
      const year = currentDate.getFullYear();
      const month = currentDate.getMonth();
      monthEl.textContent = `${year}年${month + 1}月`;
      
      grid.innerHTML = '';
      
      const dayNames = ['日', '月', '火', '水', '木', '金', '土'];
      dayNames.forEach(name => {
        const day = document.createElement('div');
        day.className = 'calendar-day header';
        day.textContent = name;
        grid.appendChild(day);
      });
      
      const firstDay = new Date(year, month, 1).getDay();
      const daysInMonth = new Date(year, month + 1, 0).getDate();
      
      for (let i = 0; i < firstDay; i++) {
        const empty = document.createElement('div');
        empty.className = 'calendar-day';
        empty.style.visibility = 'hidden';
        grid.appendChild(empty);
      }
      
      for (let day = 1; day <= daysInMonth; day++) {
        const dateStr = `${year}-${String(month + 1).padStart(2, '0')}-${String(day).padStart(2, '0')}`;
        const hasEvent = events.some(e => e.date === dateStr);
        
        const dayEl = document.createElement('div');
        dayEl.className = 'calendar-day';
        if (hasEvent) dayEl.classList.add('has-event');
        dayEl.textContent = day;
        dayEl.onclick = () => openEventModal(dateStr);
        grid.appendChild(dayEl);
      }
    }

    function changeMonth(delta) {
      currentDate.setMonth(currentDate.getMonth() + delta);
      renderCalendar();
      renderEvents();
    }

    function openEventModal(date = null) {
      const modal = document.getElementById('eventModal');
      if (date) {
        document.getElementById('eventDate').value = date;
      } else {
        document.getElementById('eventDate').value = '';
      }
      document.getElementById('eventTime').value = '';
      document.getElementById('eventTitle').value = '';
      document.getElementById('eventDescription').value = '';
      modal.classList.add('active');
    }

    function closeEventModal() {
      document.getElementById('eventModal').classList.remove('active');
    }

    function saveEvent() {
      const date = document.getElementById('eventDate').value;
      const time = document.getElementById('eventTime').value;
      const title = document.getElementById('eventTitle').value;
      const description = document.getElementById('eventDescription').value;
      
      if (!date || !title) {
        alert('日付とタイトルを入力してください');
        return;
      }
      
      events.push({
        id: Date.now(),
        date: date,
        time: time,
        title: title,
        description: description
      });
      
      events.sort((a, b) => {
        const dateA = new Date(a.date + ' ' + (a.time || '00:00'));
        const dateB = new Date(b.date + ' ' + (b.time || '00:00'));
        return dateA - dateB;
      });
      
      saveEvents();
      renderCalendar();
      renderEvents();
      closeEventModal();
    }

    function renderEvents() {
      const list = document.getElementById('eventList');
      list.innerHTML = '';
      
      const now = new Date();
      const upcomingEvents = events.filter(e => new Date(e.date) >= now);
      
      upcomingEvents.forEach(event => {
        const item = document.createElement('div');
        item.className = 'event-item';
        if (selectedItems.event.has(event.id)) item.classList.add('selected');
        
        const date = new Date(event.date);
        const dateStr = `${date.getMonth() + 1}月${date.getDate()}日`;
        const timeStr = event.time ? ` ${event.time}` : '';
        
        item.innerHTML = `
          <div class="event-checkbox" onclick="toggleEventSelection(${event.id}, event)"></div>
          <div>
            <div style="font-weight:500; margin-bottom:4px;">${event.title}</div>
            <div style="font-size:13px; color:var(--text-tertiary);">${dateStr}${timeStr}</div>
            ${event.description ? `<div style="font-size:13px; color:var(--text-secondary); margin-top:4px;">${event.description}</div>` : ''}
          </div>
        `;
        
        list.appendChild(item);
      });
      
      if (upcomingEvents.length === 0) {
        list.innerHTML = '<p style="text-align:center; color:var(--text-tertiary); padding:40px;">予定がありません</p>';
      }
    }

    // Theme functions

      localStorage.setItem('shiori_font', font);

    function applyPreset(presetName) {
      const preset = themePresets[presetName];
      applyThemeColors(preset);
      updateColorInputs(preset);
      localStorage.setItem('shiori_theme', JSON.stringify(preset));
    }

      document.documentElement.style.setProperty('--bg-primary', colors.bgPrimary);

      document.documentElement.style.setProperty('--text-secondary', colors.textSecondary);
      document.documentElement.style.setProperty('--text-tertiary', colors.textTertiary);

    }

      const colors = {
        bgPrimary: document.getElementById('bgPrimary').value,

        bgTertiary: document.getElementById('bgTertiary').value,
        accentPrimary: document.getElementById('accentPrimary').value,

      const hexInput = document.getElementById(id + 'Hex');
      const colorInput = document.getElementById(id);
      colorInput.value = hexInput.value;
      updateCustomTheme();

    function resetTheme() {
      if (!confirm('テーマをデフォルトに戻しますか?')) return;
      localStorage.removeItem('shiori_theme');
      applyPreset('default');
    }
